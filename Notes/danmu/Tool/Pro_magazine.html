      const res = await fetch(url, {
        method: 'POST',
        headers: headers,
        body: JSON.stringify(bodyData)
      });
      if (!res.ok) throw new Error('AI Interface Error: ' + res.status);
      const json = await res.json();
      if (json.choices && json.choices.length > 0) return json.choices[0].message.content.trim();
      throw new Error('AI 返回数据异常');
    }

    async function executeAIPipeline(text) {
      const customUrl = localStorage.getItem('CUSTOM_API_URL') || '';
      const customKey = localStorage.getItem('CUSTOM_API_KEY') || '';
      const customModel = localStorage.getItem('CUSTOM_MODEL') || '';
      const groqKey = localStorage.getItem('GROQ_API_KEY') || '';
      const groqModel = localStorage.getItem('GROQ_MODEL') || '';

      if (customUrl && customKey) {
        try {
          return await fetchCustomAI(text, customUrl, customKey, customModel);
        } catch (err) {
          if (groqKey) return await fetchGroq(text, groqKey, groqModel);
          throw err;
        }
      } else if (groqKey) {
        return await fetchGroq(text, groqKey, groqModel);
      } else {
        const userKey = prompt('⚠️ 当前页面未检测到 AI Key！\n为了安全，密匙只储存在您本地浏览器中。\n\n请输入您的 API Key (如 sk-xxx 或 gsk_xxx):');
        if (userKey && userKey.trim()) {
          const cleanKey = userKey.trim();
          if (cleanKey.startsWith('gsk_')) {
            localStorage.setItem('GROQ_API_KEY', cleanKey);
          } else {
            localStorage.setItem('CUSTOM_API_KEY', cleanKey);
            localStorage.setItem('CUSTOM_API_URL', 'https://api.chatanywhere.tech/v1/chat/completions');
          }
        }
        throw new Error('未设置 AI API Key');
      }
    }

    function getPurifiedFallbackHTML() {
      const clone = document.documentElement.cloneNode(true);
      const foreignSelectors = [
        'relingo-app', 'relin-highlight', 'relin-hc', '[class*="relingo"]',
        '[data-relin-paragraph]', 'grammarly-extension', '[data-grammarly-part]',
        '[class*="immersive-translate"]', '[data-immersive-translate]'
      ].join(',');
      clone.querySelectorAll(foreignSelectors).forEach(el => el.remove());
      clone.querySelectorAll('relin-highlight, relin-hc').forEach(t => {
        const frag = document.createDocumentFragment();
        while(t.firstChild) frag.appendChild(t.firstChild);
        t.parentNode.replaceChild(frag, t);
      });
      clone.querySelectorAll('script').forEach(s => {
        if (!s.src.includes('marked.min.js') && s.id !== 'core-engine') s.remove();
      });
      clone.querySelectorAll('style').forEach(s => {
        if (s.id !== 'core-style') s.remove();
      });
      const liveTAs = document.querySelectorAll('.anno-edit');
      clone.querySelectorAll('.anno-edit').forEach((ta, i) => {
        if(liveTAs[i]) ta.textContent = liveTAs[i].value.trim();
        const box = ta.closest('.anno-box');
        if (box) box.style.display = 'none';
        const view = box ? box.querySelector('.anno-view') : null;
        if (view) view.innerHTML = '';
      });
      return '<!DOCTYPE html>\n<html lang="zh-CN">\n' + clone.innerHTML + '\n</html>';
    }

    async function syncToCloud() {
      const ghToken = localStorage.getItem('LITIT_GH_TOKEN') || localStorage.getItem('GH_TOKEN');
      const ghOwner = 'moodHappy';
      const ghRepo = 'dual-read-notes';
      if (!ghToken) return;

      const pathParts = window.location.pathname.split('/');
      const fileName = pathParts.pop();
      const month = pathParts.pop();
      const year = pathParts.pop();
      if(!fileName || !month || !year) return;

      const fileRelPath = 'Summary/' + year + '/' + month + '/' + fileName;

      try {
        const getRes = await fetch('https://api.github.com/repos/' + ghOwner + '/' + ghRepo + '/contents/' + fileRelPath + '?t=' + Date.now(), {
          headers: { 'Authorization': 'Bearer ' + ghToken },
          cache: 'no-store'
        });
        let finalHTML = '';
        let sha = '';
        if (getRes.ok) {
          const fileData = await getRes.json();
          sha = fileData.sha;
          const cleanRawHTML = decodeURIComponent(escape(atob(fileData.content.replace(/\n/g, ''))));
          const parser = new DOMParser();
          const cleanDoc = parser.parseFromString(cleanRawHTML, 'text/html');

          const liveTAs = document.querySelectorAll('.anno-edit');
          const cleanTAs = cleanDoc.querySelectorAll('.anno-edit');
          liveTAs.forEach((liveTa, index) => {
            if (cleanTAs[index]) {
              const val = liveTa.value.trim();
              cleanTAs[index].textContent = val;
              const box = cleanTAs[index].closest('.anno-box');
              if (box) box.style.display = 'none';
              const view = box ? box.querySelector('.anno-view') : null;
              if (view) view.innerHTML = '';
            }
          });
          cleanDoc.querySelectorAll('.anno-toggle').forEach(t => {
            t.classList.remove('has-anno');
            const ta = t.closest('.content-wrap, .litit-highlight-block')?.querySelector('.anno-edit');
            if (ta && ta.textContent.trim()) t.classList.add('has-anno');
          });
          finalHTML = '<!DOCTYPE html>\n<html lang="zh-CN">\n' + cleanDoc.documentElement.innerHTML + '\n</html>';
        } else {
          finalHTML = getPurifiedFallbackHTML();
        }

        await fetch('https://api.github.com/repos/' + ghOwner + '/' + ghRepo + '/contents/' + fileRelPath, {
          method: 'PUT',
          headers: {
            'Authorization': 'Bearer ' + ghToken,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            message: 'Silent purified annotation update',
            content: btoa(unescape(encodeURIComponent(finalHTML))),
            sha: sha || undefined
          })
        });
        console.log('✅ [My Litit] 纯净无污染批注已静默同步至 GitHub');
      } catch(e) {
        console.warn('❌ [My Litit] 同步跳过或失败:', e);
      }
    }

    function scheduleSync() {
      if (syncTimeout) clearTimeout(syncTimeout);
      syncTimeout = setTimeout(() => { syncToCloud(); }, 3000);
    }

    function initAnnotations() {
      document.querySelectorAll('.litit-highlight-block, .content-wrap').forEach(wrap => {
        let content = wrap.querySelector('.content');
        if (!content) {
          const pText = wrap.querySelector('p');
          if (pText) {
            content = document.createElement('div');
            content.className = 'content';
            content.style.cssText = 'position:relative; margin:0 0 6px 0; color:#1a202c; font-family:Georgia, serif; line-height:1.7; font-size:15px; word-break:break-word;';
            content.innerHTML = pText.innerHTML;
            pText.parentNode.replaceChild(content, pText);
          }
        }
        if (!content) return;

        if (!content.querySelector('.anno-toggle')) {
          const toggle = document.createElement('span');
          toggle.className = 'anno-toggle';
          toggle.title = '手写批注/笔记';
          content.appendChild(toggle);
        }
        if (!content.querySelector('.ai-toggle')) {
          const aiToggle = document.createElement('span');
          aiToggle.className = 'ai-toggle';
          aiToggle.title = 'AI智能二次解析';
          content.appendChild(aiToggle);
        }

        let box = wrap.querySelector('.anno-box');
        if (!box) {
          box = document.createElement('div');
          box.className = 'anno-box';
          box.style.display = 'none';
          box.innerHTML = '<div class="anno-view markdown-body"></div><textarea class="anno-edit" style="display:none;" placeholder="在此写下笔记或使用 AI 解析..."></textarea>';
          wrap.appendChild(box);
        }

        const view = box.querySelector('.anno-view');
        const edit = box.querySelector('.anno-edit');
        const toggle = content.querySelector('.anno-toggle');
        const aiToggle = content.querySelector('.ai-toggle');

        if (edit && edit.value.trim()) {
          toggle.classList.add('has-anno');
          try {
            view.innerHTML = (typeof marked !== 'undefined') ? marked.parse(edit.value.trim()) : edit.value.trim();
          } catch(e){}
        }

        if (aiToggle) {
          aiToggle.onclick = async (e) => {
            e.preventDefault(); e.stopPropagation();
            if (aiToggle.classList.contains('loading')) return;
            if (edit && edit.value.trim() !== '') {
              const confirmOverwrite = confirm('⚠️ 已经存在批注内容。点击"确定"将使用 AI 解析覆盖现有内容。是否继续？');
              if (!confirmOverwrite) return;
            }
            const pClone = content.cloneNode(true);
            pClone.querySelectorAll('.anno-toggle, .ai-toggle').forEach(el => el.remove());
            let pText = pClone.textContent.replace(/https?:\/\/\S+/g, '').trim();
            if (!pText) return;

            aiToggle.classList.add('loading');
            try {
              const aiContent = await executeAIPipeline(pText);
              box.style.display = 'block';
              view.style.display = 'none';
              edit.style.display = 'block';
              edit.value = aiContent;
              edit.focus(); edit.blur();
            } catch (err) {
              console.error(err);
            } finally {
              aiToggle.classList.remove('loading');
            }
          };
        }

        if (toggle) {
          toggle.onclick = (e) => {
            e.preventDefault(); e.stopPropagation();
            if (box.style.display === 'block') {
              box.style.display = 'none';
            } else {
              box.style.display = 'block';
              if (!edit.value.trim()) {
                view.style.display = 'none';
                edit.style.display = 'block';
                setTimeout(() => { edit.focus(); }, 150);
              } else {
                view.style.display = 'block';
                edit.style.display = 'none';
              }
            }
          };
        }

        const triggerEdit = (e) => {
          if(e) { e.preventDefault(); e.stopPropagation(); }
          view.style.display = 'none';
          edit.style.display = 'block';
          edit.value = edit.value;
          setTimeout(() => { edit.focus(); }, 150);
        };

        view.addEventListener('dblclick', triggerEdit);
        let lastTap = 0;
        view.addEventListener('touchstart', e => {
          if (e.touches.length === 2) {
            triggerEdit(e);
          } else if (e.touches.length === 1) {
            const currentTime = new Date().getTime();
            if (currentTime - lastTap < 500 && currentTime - lastTap > 0) {
              e.preventDefault();
              box.style.display = 'none';
            }
            lastTap = currentTime;
          }
        }, {passive: false});

        if (edit && view) {
          edit.onblur = () => {
            setTimeout(() => {
              const newVal = edit.value.trim();
              try {
                view.innerHTML = newVal ? ((typeof marked !== 'undefined') ? marked.parse(newVal) : newVal) : '';
              } catch(e){}
              edit.style.display = 'none';
              if (newVal) {
                view.style.display = 'block';
                toggle.classList.add('has-anno');
              } else {
                view.style.display = 'none';
                box.style.display = 'none';
                toggle.classList.remove('has-anno');
              }
              if (edit.getAttribute('data-old-val') !== newVal) {
                edit.setAttribute('data-old-val', newVal);
                scheduleSync();
              }
            }, 150);
          };
          edit.setAttribute('data-old-val', edit.value.trim());
        }
      });
    }

    window.addEventListener('DOMContentLoaded', () => {
      initProgressBar();
      initAnnotations();
    });

    window.addEventListener('load', () => {
      initAnnotations();
    });
